//Write a program to check if a number is a perfect square.

import java.util.Scanner;

public class Q1{
  public static boolean isPerfectSquare(int num){
     int i = 1;
     while (i*i < num) {
        i++;
     }
     if (i * i == num) {
        return true;
     }
     return false;
  } 
  public static void main(String[] args){
     Scanner sc=new Scanner(System.in);
     System.out.println("Enter number ");
     int num=sc.nextInt();
     if(isPerfectSquare(num)){
      System.out.println("number "+num+" is perfect squalre");
     }
     else{
         System.out.println("number "+num+" is not perfect squalre");
     }
  }
}


//Write a program to merge two sorted arrays into a single sorted array.

import java.util.Arrays;

public class Q2 {
   public static void main(String[] args) {
      int[] arr1={1,2,3,4,5,6};
      int[] arr2={1,3,4,6,7,8};
      int len=arr1.length+arr2.length;
      int[] ans=new int[len];

      int i=0;
      int j=0;
      int k=0;
      while(i<arr1.length && j<arr2.length){
        if(arr1[i]<arr2[j]){
           ans[k++]=arr1[i++];
        }
        else{
           ans[k++] = arr2[j++];
        }
      }

      while (j < arr2.length) {
         ans[k++] = arr2[j++];
      }

      while (i < arr1.length) {
         ans[k++] = arr1[i++];
      }

      System.out.println(Arrays.toString(ans));
   }
}
